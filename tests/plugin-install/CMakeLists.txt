cmake_minimum_required(VERSION 3.20)
project(plugin_install_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
get_filename_component(ACCUSHOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(plugin_install_test plugin_install_test.cpp)
target_include_directories(plugin_install_test PRIVATE "${ACCUSHOT_ROOT}/src")
target_link_libraries(plugin_install_test PRIVATE Qt6::Core Qt6::Test)
target_compile_options(plugin_install_test PRIVATE /utf-8 /W4)
target_compile_definitions(plugin_install_test PRIVATE INSTALLER_PATH="${ACCUSHOT_ROOT}/scripts/local_plugins/install.py")
include("${ACCUSHOT_ROOT}/scripts/bundle_plugin_tools.cmake")
accushot_bundle_plugin_tools(plugin_install_test "${ACCUSHOT_ROOT}")
enable_testing()
add_test(NAME plugin_install_test COMMAND plugin_install_test)
add_executable(python_installer_test python_installer_test.cpp
    "${ACCUSHOT_ROOT}/src/core/plugin/python_installer.cpp")
target_include_directories(python_installer_test PRIVATE "${ACCUSHOT_ROOT}/src")
target_link_libraries(python_installer_test PRIVATE Qt6::Core Qt6::Network Qt6::Test)
target_compile_options(python_installer_test PRIVATE /utf-8 /W4)
add_test(NAME python_installer_test COMMAND python_installer_test)
