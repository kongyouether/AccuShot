cmake_minimum_required(VERSION 3.20)
project(accushot_ocr_link LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg Widgets Network Test)
get_filename_component(ACCUSHOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(GLOB_RECURSE SOURCES "${ACCUSHOT_ROOT}/src/*.cpp" "${ACCUSHOT_ROOT}/src/*.h")
list(REMOVE_ITEM SOURCES "${ACCUSHOT_ROOT}/src/main.cpp")
add_executable(ocr_link ocr_link.cpp ${SOURCES} "${ACCUSHOT_ROOT}/resources/resources.qrc")
target_include_directories(ocr_link PRIVATE "${ACCUSHOT_ROOT}/src")
target_link_libraries(ocr_link PRIVATE Qt6::Core Qt6::Gui Qt6::Svg Qt6::Widgets Qt6::Network Qt6::Test user32 gdi32 ole32 uiautomationcore dwmapi windowsapp runtimeobject advapi32 dbghelp)
target_compile_options(ocr_link PRIVATE /utf-8 /permissive- /W4)
enable_testing()
add_test(NAME ocr_link COMMAND ocr_link)
set_tests_properties(ocr_link PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
