cmake_minimum_required(VERSION 3.20)
project(accushot_result_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg Widgets Network Test)
get_filename_component(ACCUSHOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(GLOB_RECURSE SOURCES "${ACCUSHOT_ROOT}/src/*.cpp" "${ACCUSHOT_ROOT}/src/*.h")
list(REMOVE_ITEM SOURCES "${ACCUSHOT_ROOT}/src/main.cpp")
add_executable(result_ui result_ui.cpp ${SOURCES} "${ACCUSHOT_ROOT}/resources/resources.qrc")
include("${ACCUSHOT_ROOT}/scripts/bundle_plugin_tools.cmake")
accushot_bundle_plugin_tools(result_ui "${ACCUSHOT_ROOT}")
target_include_directories(result_ui PRIVATE "${ACCUSHOT_ROOT}/src")
target_link_libraries(result_ui PRIVATE Qt6::Core Qt6::Gui Qt6::Svg Qt6::Widgets Qt6::Network Qt6::Test user32 gdi32 ole32 uiautomationcore dwmapi windowsapp runtimeobject advapi32 dbghelp)
target_compile_options(result_ui PRIVATE /utf-8 /permissive- /W4)
enable_testing()
add_test(NAME result_ui COMMAND result_ui)
set_tests_properties(result_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(ai_probe ../ai-probe/main.cpp "${ACCUSHOT_ROOT}/src/core/ai/ai_service.cpp")
target_include_directories(ai_probe PRIVATE "${ACCUSHOT_ROOT}/src")
target_link_libraries(ai_probe PRIVATE Qt6::Core Qt6::Gui Qt6::Svg Qt6::Network advapi32)
target_compile_options(ai_probe PRIVATE /utf-8 /permissive-)
add_executable(native_snap_probe ../ai-probe/native_snap_probe.cpp "${ACCUSHOT_ROOT}/src/core/smart_snapper.cpp" "${ACCUSHOT_ROOT}/src/core/native_snapper.cpp")
target_include_directories(native_snap_probe PRIVATE "${ACCUSHOT_ROOT}/src")
target_link_libraries(native_snap_probe PRIVATE Qt6::Core Qt6::Gui Qt6::Svg Qt6::Widgets user32 dwmapi ole32 uiautomationcore)
target_compile_options(native_snap_probe PRIVATE /utf-8 /permissive-)
