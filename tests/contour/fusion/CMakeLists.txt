cmake_minimum_required(VERSION 3.20)
project(fusion_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
get_filename_component(ROOT "../../.." ABSOLUTE)
add_executable(fusion_probe ../fusion_probe.cpp "${ROOT}/src/core/smart_snapper.cpp" "${ROOT}/src/core/native_snapper.cpp" "${ROOT}/src/core/screen_capturer.cpp")
target_include_directories(fusion_probe PRIVATE "${ROOT}/src/core")
target_link_libraries(fusion_probe PRIVATE Qt6::Core Qt6::Gui user32 dwmapi ole32 oleaut32 uiautomationcore)
target_compile_options(fusion_probe PRIVATE /utf-8 /O2)
