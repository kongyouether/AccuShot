cmake_minimum_required(VERSION 3.20)
project(contour_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
get_filename_component(ROOT "../.." ABSOLUTE)
foreach(variant IN ITEMS baseline current)
  if(variant STREQUAL baseline)
    set(SNAP baseline.cpp)
  else()
    set(SNAP "${ROOT}/src/core/smart_snapper.cpp")
  endif()
  add_executable(contour_${variant} probe.cpp ${SNAP} "${ROOT}/src/core/native_snapper.cpp")
  target_include_directories(contour_${variant} PRIVATE "${ROOT}/src/core")
  target_link_libraries(contour_${variant} PRIVATE Qt6::Core Qt6::Gui user32 dwmapi ole32 oleaut32 uiautomationcore)
  target_compile_options(contour_${variant} PRIVATE /utf-8 /O2)
endforeach()
add_executable(contour_regression regression.cpp "${ROOT}/src/core/smart_snapper.cpp" "${ROOT}/src/core/native_snapper.cpp")
target_include_directories(contour_regression PRIVATE "${ROOT}/src/core")
target_link_libraries(contour_regression PRIVATE Qt6::Core Qt6::Gui user32 dwmapi ole32 oleaut32 uiautomationcore)
target_compile_options(contour_regression PRIVATE /utf-8 /O2)
add_executable(fusion_regression fusion_regression.cpp "${ROOT}/src/core/smart_snapper.cpp" "${ROOT}/src/core/native_snapper.cpp")
target_include_directories(fusion_regression PRIVATE "${ROOT}/src/core")
target_link_libraries(fusion_regression PRIVATE Qt6::Core Qt6::Gui user32 dwmapi ole32 oleaut32 uiautomationcore)
target_compile_options(fusion_regression PRIVATE /utf-8 /O2)
