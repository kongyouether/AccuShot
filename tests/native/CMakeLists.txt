cmake_minimum_required(VERSION 3.20)
project(evan_native_e2e LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg Widgets Network Test)
get_filename_component(EVAN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(GLOB_RECURSE EVAN_SOURCES CONFIGURE_DEPENDS "${EVAN_ROOT}/src/*.cpp" "${EVAN_ROOT}/src/*.h")
list(REMOVE_ITEM EVAN_SOURCES "${EVAN_ROOT}/src/main.cpp")
add_executable(native_e2e ../native_e2e.cpp ${EVAN_SOURCES} "${EVAN_ROOT}/resources/resources.qrc")
target_include_directories(native_e2e PRIVATE "${EVAN_ROOT}/src" "${EVAN_ROOT}/src/core" "${EVAN_ROOT}/src/ui" "${EVAN_ROOT}/src/platform")
target_link_libraries(native_e2e PRIVATE Qt6::Core Qt6::Gui Qt6::Svg Qt6::Widgets Qt6::Network Qt6::Test user32 gdi32 dwmapi windowsapp runtimeobject advapi32 dbghelp)
if(MSVC)
    target_compile_options(native_e2e PRIVATE /utf-8 /permissive- /W4)
endif()
enable_testing()
add_executable(diagnostics_probe ../diagnostics_probe.cpp "${EVAN_ROOT}/src/core/diagnostics.cpp")
target_include_directories(diagnostics_probe PRIVATE "${EVAN_ROOT}/src")
target_link_libraries(diagnostics_probe PRIVATE Qt6::Core dbghelp)
if(MSVC)
    target_compile_options(diagnostics_probe PRIVATE /utf-8)
endif()
add_test(NAME native_e2e COMMAND native_e2e)
set_tests_properties(native_e2e PROPERTIES TIMEOUT 60)
